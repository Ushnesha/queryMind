\end{itemize}

\item Tie — Unanswerable: correct refusal: 14/14 vs 14/14
\item Tie — False answers on unanswerable: 0/14 vs 0/14
\end{itemize}
\par\textit{What it costs}
\begin{itemize}
\item 3.2$\times$ the LLM calls of b1

\item p50 latency on answered questions 11.9 s vs 4.2 s
\end{itemize}
